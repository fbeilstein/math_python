\section*{Problem 1: Prime Fields GF(p)}
\textbf{Mathematical part:} Implement modular arithmetic for a prime field $GF(p)$. Ensure division handles the modular inverse correctly using Fermat's Little Theorem or the Extended Euclidean Algorithm. \\
\textbf{Implementation part:} Implement \lstinline|gfp_add|, \lstinline|gfp_sub|, \lstinline|gfp_mul|, and \lstinline|gfp_div|.

\section*{Problem 2: Primitive Polynomials}
\textbf{Mathematical part:} A polynomial is primitive over $GF(p)$ if it is irreducible and its roots generate the multiplicative group of the extension field. \\
\textbf{Implementation part:} Implement \lstinline|gfp_poly_divide| and \lstinline|is_primitive| to find valid generators for $GF(p^n)$.

\section*{Problem 3: Extension Fields GF(p\textsuperscript{n}) Tables}
\textbf{Mathematical part:} To quickly multiply in $GF(p^n)$, we construct Exponential and Logarithm tables using a chosen primitive polynomial. \\
\textbf{Implementation part:} Implement \lstinline|generate_gfpn_tables|.

\section*{Problem 4: Polynomial Arithmetic in GF(p\textsuperscript{n})}
\textbf{Implementation part:} Implement polynomial multiplication and division over $GF(p^n)$ using the tables you just generated.

\section*{Problem 5: Reed-Solomon Encoding}
\textbf{Mathematical part:} The Reed-Solomon generator polynomial has roots $\alpha^0, \alpha^1, \dots, \alpha^{2t-1}$. The encoded message is the data shifted by $x^{2t}$ plus the remainder when divided by the generator. \\
\textbf{Implementation part:} Implement \lstinline|get_generator_poly| and \lstinline|encode_text|.

\section*{Problems 6-8: Reed-Solomon Decoding}
\textbf{Mathematical part:} Implement the decoding pipeline:
\begin{enumerate}
    \item \textbf{Syndromes (L6):} Evaluate the received polynomial at the roots of the generator.
    \item \textbf{Berlekamp-Massey (L7):} Solve the LFSR to find the Error Locator Polynomial $\Lambda(x)$.
    \item \textbf{Chien Search \& Forney (L8):} Find the roots of $\Lambda(x)$ and calculate the error magnitudes to recover the corrupted bytes.
\end{enumerate}

\section*{Problem 9: The QR Matrix}
\textbf{Implementation part:} Route your robustly encoded Reed-Solomon bitstream into the physical 2D matrix of a QR code, avoiding the finder patterns. Run \lstinline|qr_code_demo.py| to test your final matrix with a real smartphone camera!
